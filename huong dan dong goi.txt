Tạo các bảng thủ công

-- Tạo bảng User (đặt trong dấu ngoặc kép vì user là từ khóa)
CREATE TABLE "User" (
    id SERIAL PRIMARY KEY,
    student_id VARCHAR(20) UNIQUE NOT NULL,
    name VARCHAR(100),
    class_name VARCHAR(50),
    khoa_vien VARCHAR(100),
    password_hash VARCHAR(128) NOT NULL,
    plain_password VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tạo bảng Computer
CREATE TABLE Computer (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50),
    mac_address VARCHAR(20),
    ip_address VARCHAR(15),
    status VARCHAR(20) DEFAULT 'offline',
    last_active TIMESTAMP,
    api_port INTEGER DEFAULT 5001
);

-- Tạo bảng Session
CREATE TABLE Session (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES "User"(id),
    computer_id INTEGER NOT NULL REFERENCES Computer(id),
    login_time TIMESTAMP NOT NULL,
    logout_time TIMESTAMP,
    duration INTEGER
);

-- Tạo bảng Advertisement
CREATE TABLE Advertisement (
    id SERIAL PRIMARY KEY,
    image_url VARCHAR(500) NOT NULL,
    link VARCHAR(500) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);

-- Tạo các index để tối ưu hiệu năng
CREATE INDEX idx_session_user_id ON Session(user_id);
CREATE INDEX idx_session_computer_id ON Session(computer_id);
CREATE INDEX idx_advertisement_active ON Advertisement(is_active);
